# This file lists the Python packages needed for the project.
# Install them with:
# pip install -r requirements.txt

# Ultralytics provides YOLOv8.
# It is used for training, validation, and prediction.
ultralytics

# OpenCV is used for reading images, showing images, and using the webcam.
opencv-python

# NumPy is used for basic numerical operations.
# OpenCV and YOLO also work with NumPy arrays.
numpy

# Roboflow is optional, but useful if you want to download datasets directly.
roboflow

# Pandas is used by run_experiments.py to read YOLO results.csv files
# and create the final comparison table.
pandas

# Matplotlib is used by run_experiments.py to create comparison_plot.png.
matplotlib
